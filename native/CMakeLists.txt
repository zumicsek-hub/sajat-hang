cmake_minimum_required(VERSION 3.20)
project(sajat_hang_dsp LANGUAGES CXX)
enable_testing()
add_executable(dsp_tests tests/processor_test.cpp)
target_compile_features(dsp_tests PRIVATE cxx_std_17)
target_include_directories(dsp_tests PRIVATE ../packages/voice_audio/native/include)
if(MSVC)
  target_compile_options(dsp_tests PRIVATE /W4 /WX)
else()
  target_compile_options(dsp_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME dsp_safety COMMAND dsp_tests)
